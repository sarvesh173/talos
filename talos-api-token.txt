import time, requests

BASE = "http://192.168.31.209:8000"
H = {"Authorization": "Bearer talos_«redacted:sk_…»"}

r = requests.post(f"{BASE}/api/opencode/sessions", headers=H, json={})
sid = r.json()["session"]["id"]

requests.post(
    f"{BASE}/api/opencode/sessions/{sid}/prompt",
    headers=H, json={"text": "Say only: PONG"}, timeout=30,
)

deadline = time.time() + 120
while time.time() < deadline:
    time.sleep(5)
    msgs = requests.get(
        f"{BASE}/api/opencode/sessions/{sid}/messages",
        headers=H, params={"limit": 50}, timeout=20,
    ).json()["messages"]
    for m in msgs:
        if m["role"] == "assistant" and m["text"].strip():
            print(m["text"])
            raise SystemExit
print("timeout")
